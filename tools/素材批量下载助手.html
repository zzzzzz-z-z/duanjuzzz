创量「素材筛选」页 · 批量下载助手 书签 v4

安装（任选一种）

方式一（推荐）：把下面蓝色按钮 拖到浏览器书签栏（需先打开书签栏：Ctrl+Shift+B）。

素材批量下载助手 v4

方式二：点下面按钮复制书签链接，手动新建书签粘贴。

v4 修复：名字按“行”分隔

剧名里自带中文逗号（如 「唢呐修仙，一曲炸翻修仙界第七季」）是一整个剧名，之前会被错误拆成两个。v4 起只按换行分行，一个剧名一整行，逗号保留。

功能汇总

功能说明
名字输入每行一个剧名，可批量粘贴；一行 = 一个剧
时长过滤跳过时长 <100 或 >1000 秒的素材，凑满 10 个合格素材（范围可改）
加载处理未加载的先点下载触发 → 等 30 秒 → 点查询 → 最多 3 轮 → 全部已加载后逐个下载
面板可拖动 / 关闭 × / 收起 — / 停止 / 诊断 / 清空日志
日志绿√成功 / 红×失败 / 橙提示

使用步骤

  1. 打开创量投放平台 → 素材筛选 页并登录。
  2. 点书签，出现面板。
  3. 文本框里每行一个剧名，点开始。脚本自动：输入名字→查询→扫描并跳过时长不合格的→凑满10个→触发加载→下载→下一个名字。
  4. 日志绿色√成功 / 红色×失败 / 橙色提示。

排查“只下载前4个”：先用「诊断」

在素材筛选页搜出结果后点「诊断」，把日志整段复制发给助手。重点看：

  • ① .el-table__row 数量：若只有 4~6 个，说明表格是虚拟滚动（DOM 只渲染可见行），需要改成“逐行滚动处理”逻辑。
  • ④ 滚动测试：滚动后行数是否变化，判断是否懒加载。
  • ③ 前12行：会附带每行时长，确认时长读取正确。
注意：若浏览器弹「此网站正在尝试下载多个文件」，点允许。处理期间保持页面在前台，不要刷新/切走。
说明：下载按平台口径执行（扣今日下载量额度）。若合格素材不足 10 个，按实际数量下载。

源码（v4，可自行修改）

// =============================================================
// 创量投放平台「素材筛选」页 · 批量下载助手 书签 v4
// -------------------------------------------------------------
// v4 新增：名字按行分隔，剧名内的逗号不再拆分
// v3 新增：时长过滤
//  按顺序扫描素材，跳过 时长<MIN_DUR 或 >MAX_DUR 的（默认100~1000秒），
//  继续往后找，直到凑满 COUNT 个合格素材再下载。
// v2 已有：
//  1) 面板可拖动（按住标题栏拖动）
//  2) 面板可关闭（×）、可收起（—）、可停止（停止）
//  3) 「诊断」按钮：输出真实 DOM 行数/滚动容器/前12行状态
//  4) 点击每个下载前先滚动到可视区
//  5) 查询后先回到顶部再取前 N 个，保证顺序
// =============================================================
(() => {
  'use strict';

  // ===== 可调参数（面板下拉也可改） =====
  // 每个名字取前 N 个素材
  let COUNT = 10;
  // 触发加载后等待毫秒数
  let WAIT_MS = 30000;
  // 最多“等待+查询”轮次
  let MAX_ROUNDS = 3;
  // 时长过滤：只保留 [MIN_DUR, MAX_DUR] 秒内的素材（面板可改）
  let MIN_DUR = 100;
  let MAX_DUR = 1000;
  // 触发加载时连续点击间隔(ms)
  const LOAD_GAP = 500;
  // 正式下载时连续点击间隔(ms)
  const DOWNLOAD_GAP = 2000;

  // ===== 页面元素选择器 =====
  const INP_SEL = 'input.el-input__inner[placeholder="搜索产品名/短剧名/公司名/文案/落地页"]';
  const ROW_SEL = '.el-table__row';

  const $ = (s, c) => (c || document).querySelector(s);
  const $$ = (s, c) => Array.prototype.slice.call((c || document).querySelectorAll(s));

  const sleep = (ms) => {
    const t0 = Date.now();
    return new Promise((resolve) => {
      const loop = () => (Date.now() - t0 >= ms ? resolve() : setTimeout(loop, 200));
      loop();
    });
  };

  // 停止标记
  let STOPPED = false;
  const stopIfNeeded = async () => {
    if (STOPPED) throw new Error('已手动停止');
  };

  // ===== 浮层面板 =====
  let panel = document.getElementById('dl-assist-panel');
  if (!panel) {
    panel = document.createElement('div');
    panel.id = 'dl-assist-panel';
    panel.style.cssText = 'position:fixed;top:10px;left:10px;z-index:2147483647;width:440px;max-height:82vh;background:#fff;border:1px solid #c6c9cf;border-radius:8px;box-shadow:0 4px 20px rgba(0,0,0,.25);display:flex;flex-direction:column;font:12px/1.7 "Microsoft YaHei",Arial,sans-serif;color:#333;';
    panel.innerHTML =
      '<div id="dl-head" style="display:flex;align-items:center;padding:6px 10px;background:#2f6bff;color:#fff;border-radius:8px 8px 0 0;font-weight:bold;cursor:move;user-select:none;">' +
      '<span style="flex:1;">素材批量下载助手 v4</span>' +
      '<button id="dl-min" title="收起/展开" style="margin-left:6px;background:transparent;border:0;color:#fff;cursor:pointer;font-size:14px;">—</button>' +
      '<button id="dl-close" title="关闭" style="margin-left:4px;background:transparent;border:0;color:#fff;cursor:pointer;font-size:16px;">×</button>' +
      '</div>' +
      '<div id="dl-main">' +
      '<div style="padding:8px 12px;border-bottom:1px solid #eee;">' +
      '<textarea id="dl-names" style="width:100%;height:66px;box-sizing:border-box;resize:vertical;" placeholder="每行一个剧名，可批量粘贴"></textarea>' +
      '<div style="margin-top:6px;display:flex;align-items:center;flex-wrap:wrap;gap:6px;">' +
      '每名前 <select id="dl-count"><option>5</option><option selected>10</option><option>20</option></select> 个' +
      ' 等待 <select id="dl-wait"><option>15</option><option selected>30</option><option>45</option><option>60</option></select> 秒' +
      ' 最多 <select id="dl-rounds"><option>1</option><option selected>3</option><option>5</option></select> 轮' +
      ' 时长 <input id="dl-durmin" type="number" style="width:52px;" value="100">~<input id="dl-durmax" type="number" style="width:52px;" value="1000"> 秒' +
      '</div>' +
      '<div style="margin-top:8px;display:flex;gap:6px;flex-wrap:wrap;">' +
      '<button id="dl-start" style="padding:5px 16px;background:#2f6bff;color:#fff;border:0;border-radius:4px;cursor:pointer;">开始</button>' +
      '<button id="dl-stop" style="padding:5px 16px;background:#d93025;color:#fff;border:0;border-radius:4px;cursor:pointer;">停止</button>' +
      '<button id="dl-diag" style="padding:5px 14px;background:#fff;color:#333;border:1px solid #c6c9cf;border-radius:4px;cursor:pointer;">诊断</button>' +
      '<button id="dl-clear" style="padding:5px 14px;background:#fff;color:#333;border:1px solid #c6c9cf;border-radius:4px;cursor:pointer;">清空日志</button>' +
      '</div></div>' +
      '<div id="dl-log" style="flex:1;overflow:auto;padding:8px 12px;min-height:120px;max-height:48vh;"></div>' +
      '<div style="padding:6px 12px;border-top:1px solid #eee;color:#999;">状态：<span id="dl-status">待机</span></div>' +
      '</div>';
    document.body.appendChild(panel);
  } else {
    panel.style.display = 'block';
  }
  const mainBox = $('#dl-main', panel);
  const logbox = $('#dl-log', panel);
  const statusEl = $('#dl-status', panel);
  const setStatus = (t) => { statusEl.textContent = t; };
  const log = (msg, cls) => {
    const d = document.createElement('div');
    let color = '#555';
    if (cls === 'ok') color = '#0a8f2b';
    else if (cls === 'err') color = '#d93025';
    else if (cls === 'warn') color = '#e37400';
    d.style.cssText = 'padding:2px 0;color:' + color + ';word-break:break-all;';
    d.innerHTML = msg;
    logbox.appendChild(d);
    logbox.scrollTop = logbox.scrollHeight;
  };

  // ===== 面板交互：拖动 / 关闭 / 收起 =====
  const head = $('#dl-head', panel);
  let dragging = false, ox = 0, oy = 0;
  head.addEventListener('mousedown', (e) => {
    if (e.target && e.target.closest && e.target.closest('button')) return;
    dragging = true;
    ox = e.clientX - panel.offsetLeft;
    oy = e.clientY - panel.offsetTop;
    e.preventDefault();
  });
  document.addEventListener('mousemove', (e) => {
    if (!dragging) return;
    let x = e.clientX - ox;
    let y = e.clientY - oy;
    x = Math.max(0, Math.min(x, window.innerWidth - panel.offsetWidth));
    y = Math.max(0, Math.min(y, window.innerHeight - 30));
    panel.style.left = x + 'px';
    panel.style.top = y + 'px';
  });
  document.addEventListener('mouseup', () => { dragging = false; });
  $('#dl-close', panel).onclick = () => { panel.style.display = 'none'; };
  $('#dl-min', panel).onclick = () => {
    const hidden = mainBox.style.display === 'none';
    mainBox.style.display = hidden ? '' : 'none';
    $('#dl-min', panel).textContent = hidden ? '—' : '□';
  };

  // ===== 核心操作 =====
  const setSearch = (name) => {
    const inp = $(INP_SEL);
    if (!inp) throw new Error('未找到搜索框');
    try { inp.scrollIntoView({ block: 'center' }); } catch (e) {}
    const setter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set;
    setter.call(inp, name);
    inp.dispatchEvent(new Event('input', { bubbles: true }));
    inp.dispatchEvent(new Event('change', { bubbles: true }));
    if (inp.value !== name) {
      setter.call(inp, name);
      inp.dispatchEvent(new Event('input', { bubbles: true }));
    }
  };

  const clickQuery = () => {
    const bs = $$('button');
    let b = bs.find((x) => (x.textContent || '').trim() === '查询' && /120px/.test(x.getAttribute('style') || ''));
    if (!b) b = bs.find((x) => (x.textContent || '').trim() === '查询');
    if (!b) throw new Error('未找到查询按钮');
    b.click();
  };

  const rowStatus = (row) => {
    const t = $('.el-tag', row);
    if (!t) return 'unknown';
    const txt = t.textContent.trim();
    if (txt.indexOf('已加载') > -1) return 'loaded';
    if (txt.indexOf('未加载') > -1) return 'notloaded';
    return 'unknown';
  };

  const clickDownload = async (row) => {
    try { row.scrollIntoView({ block: 'center' }); } catch (e) {}
    await sleep(300);
    const b = $$('button', row).find((x) => (x.textContent || '').indexOf('下载') > -1);
    if (!b) throw new Error('行内未找到下载按钮');
    b.click();
  };

  const scrollToBottom = () => {
    window.scrollTo(0, document.body.scrollHeight);
    $$('.el-table__body-wrapper').forEach((w) => { w.scrollTop = w.scrollHeight; });
  };
  const scrollToTop = () => {
    window.scrollTo(0, 0);
    $$('.el-table__body-wrapper').forEach((w) => { w.scrollTop = 0; });
  };

  // 从行文本里取时长（秒），格式如“时长: 371秒”
  const extractDuration = (row) => {
    const m = (row.textContent || '').match(/时长[:：]\s*(\d+)\s*秒/);
    return m ? parseInt(m[1], 10) : null;
  };
  const durOk = (dur) => dur !== null && dur >= MIN_DUR && dur <= MAX_DUR;

  // 按顺序扫描行，跳过时长不合格的，凑满 limit 个合格素材（必要时滚动加载更多）
  const collectValid = async (limit) => {
    const valid = [];
    let prev = -1, same = 0;
    for (let i = 0; i < 30; i++) {
      const rows = $$(ROW_SEL);
      for (const r of rows) {
        if (valid.length >= limit) break;
        if (valid.indexOf(r) > -1) continue;
        if (durOk(extractDuration(r))) valid.push(r);
      }
      if (valid.length >= limit) break;
      const cur = rows.length;
      if (cur === prev) { same++; if (same >= 4) break; } else same = 0;
      prev = cur;
      scrollToBottom();
      await sleep(600);
    }
    scrollToTop();
    await sleep(300);
    return valid;
  };

  // 等待查询结果：行数稳定即认为加载完成
  const waitResults = async () => {
    await sleep(1200);
    const t0 = Date.now();
    let last = -1, stable = 0, zero = 0;
    while (Date.now() - t0 < 15000) {
      const n = $$(ROW_SEL).length;
      if (n === 0) { zero++; if (zero >= 10) break; }
      else {
        zero = 0;
        if (n === last) { stable++; if (stable >= 2) break; } else stable = 0;
        last = n;
      }
      await sleep(400);
    }
    await sleep(800);
  };

  // ===== 诊断：输出真实 DOM 情况 =====
  const diagnose = async () => {
    log('—— 诊断开始 ——', 'warn');
    const rows0 = $$(ROW_SEL);
    log('① .el-table__row 数量：' + rows0.length, 'info');
    const alts = ['tbody tr', '.el-table__body tr', '.el-table tbody tr', '.el-table__row'];
    alts.forEach((s) => {
      const n = $$(s).length;
      if (n > 0) log('   备选选择器 ' + s + ' → ' + n + ' 个', 'info');
    });
    log('② window: 可视高 ' + window.innerHeight + '，页面总高 ' + document.body.scrollHeight + '，已滚 ' + window.pageYOffset, 'info');
    const boxes = $$('.el-table__body-wrapper');
    boxes.forEach((w, i) => {
      log('   表格滚动容器' + i + ': clientHeight=' + w.clientHeight + ' scrollHeight=' + w.scrollHeight + ' scrollTop=' + w.scrollTop, 'info');
    });
    log('③ 前 12 行状态（DOM 里的）：', 'info');
    rows0.slice(0, 12).forEach((r, i) => {
      const st = rowStatus(r);
      const dur = extractDuration(r);
      const txt = (r.textContent || '').trim().replace(/\s+/g, ' ').slice(0, 45);
      log('   行' + i + ' [' + st + '] 时长' + (dur === null ? '?' : dur) + '秒 ' + txt, st === 'loaded' ? 'ok' : (st === 'notloaded' ? 'err' : 'info'));
    });
    log('④ 滚动测试：滚到底部后行数变化（判断是否虚拟滚动/懒加载）...', 'warn');
    const before = $$(ROW_SEL).length;
    scrollToBottom();
    await sleep(1500);
    const after = $$(ROW_SEL).length;
    log('   滚动前 ' + before + ' → 滚动后 ' + after, after > before ? 'ok' : 'info');
    scrollToTop();
    await sleep(800);
    const topN = $$(ROW_SEL).length;
    log('   回到顶部后 DOM 行数：' + topN, topN >= COUNT ? 'ok' : 'err');
    log('—— 诊断结束（把以上内容贴给助手）——', 'warn');
  };

  // ===== 单个名字处理 =====
  const processOne = async (name) => {
    setSearch(name);
    await sleep(300);
    clickQuery();
    await waitResults();
    let targets = await collectValid(COUNT);
    if (targets.length === 0) { log('× 未找到「' + name + '」的合格素材（时长 ' + MIN_DUR + '~' + MAX_DUR + ' 秒）', 'err'); return; }
    log('▶ 扫描得到 ' + targets.length + ' 个合格素材（时长 ' + MIN_DUR + '~' + MAX_DUR + ' 秒）', 'info');
    if (targets.length < COUNT) log('⚠ 合格素材不足 ' + COUNT + ' 个，按 ' + targets.length + ' 个处理', 'warn');
    let allLoaded = false;
    for (let rnd = 1; rnd <= MAX_ROUNDS; rnd++) {
      await stopIfNeeded();
      targets = await collectValid(COUNT);
      const bad = targets.filter((r) => rowStatus(r) === 'notloaded');
      if (bad.length === 0) { allLoaded = true; log('√ ' + targets.length + ' 个合格素材已全部加载', 'ok'); break; }
      log('第 ' + rnd + ' 轮：' + bad.length + ' 个未加载，点击下载触发加载，等待 ' + Math.round(WAIT_MS / 1000) + ' 秒...', 'warn');
      for (const r of bad) { try { await clickDownload(r); } catch (e) {} await sleep(LOAD_GAP); await stopIfNeeded(); }
      await sleep(WAIT_MS);
      clickQuery();
      await waitResults();
    }
    if (!allLoaded) log('× 已达最大轮次，仍有未加载', 'err');
    targets = await collectValid(COUNT);
    const ok = targets.filter((r) => rowStatus(r) === 'loaded');
    const fail = targets.filter((r) => rowStatus(r) !== 'loaded');
    for (const r of ok) { try { await clickDownload(r); } catch (e) { log('× 点击下载失败', 'err'); } await sleep(DOWNLOAD_GAP); await stopIfNeeded(); }
    log('√ 下载完成：' + ok.length + '/' + targets.length + (fail.length ? '（' + fail.length + ' 个未加载/不合格跳过）' : ''), 'ok');
  };

  // ===== 主流程 =====
  const main = async () => {
    STOPPED = false;
    const names = $('#dl-names', panel).value.split(/[\r\n]+/).map((s) => s.trim()).filter(Boolean);
    if (!names.length) { log('× 请先输入名字', 'err'); return; }
    COUNT = parseInt($('#dl-count', panel).value, 10) || 10;
    WAIT_MS = (parseInt($('#dl-wait', panel).value, 10) || 30) * 1000;
    MAX_ROUNDS = parseInt($('#dl-rounds', panel).value, 10) || 3;
    MIN_DUR = parseInt($('#dl-durmin', panel).value, 10);
    MAX_DUR = parseInt($('#dl-durmax', panel).value, 10);
    if (!(MIN_DUR >= 0)) MIN_DUR = 100;
    if (!(MAX_DUR > MIN_DUR)) MAX_DUR = 1000;
    log('开始批量处理，共 ' + names.length + ' 个名字（时长过滤 ' + MIN_DUR + '~' + MAX_DUR + ' 秒）', 'info');
    for (let i = 0; i < names.length; i++) {
      if (STOPPED) { log('已停止，剩余名字未处理', 'err'); break; }
      setStatus('处理中 ' + (i + 1) + '/' + names.length + '：' + names[i]);
      log('════ [' + (i + 1) + '/' + names.length + '] ' + names[i] + ' ════', 'info');
      try { await processOne(names[i]); } catch (e) { log('× 处理失败：' + (e && e.message || e), 'err'); }
    }
    setStatus('全部完成');
    log('√ 全部处理完成', 'ok');
  };

  // ===== 按钮绑定 =====
  $('#dl-start', panel).onclick = main;
  $('#dl-stop', panel).onclick = () => { STOPPED = true; log('已请求停止（当前步骤完成后停止）', 'warn'); };
  $('#dl-diag', panel).onclick = () => { diagnose(); };
  $('#dl-clear', panel).onclick = () => { logbox.innerHTML = ''; };
  log('面板已就绪 v4：可拖动/关闭/收起/停止；按时长 ' + MIN_DUR + '~' + MAX_DUR + ' 秒过滤凑满 ' + COUNT + ' 个。', 'info');
})();
